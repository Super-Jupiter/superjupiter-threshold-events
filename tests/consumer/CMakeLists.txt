cmake_minimum_required(VERSION 3.16)
project(SuperJupiterThresholdEventsConsumer LANGUAGES CXX)
if(DEFINED SUPERJUPITER_THRESHOLD_EVENTS_SOURCE_DIR)
  add_subdirectory("${SUPERJUPITER_THRESHOLD_EVENTS_SOURCE_DIR}" threshold-events)
  if(TARGET superjupiter-threshold-events-tests OR TARGET superjupiter-threshold-events-replay)
    message(FATAL_ERROR "A consumer unexpectedly builds library developer tools")
  endif()
else()
  find_package(SuperJupiterThresholdEvents 0.1 CONFIG REQUIRED)
endif()
add_executable(superjupiter-threshold-events-consumer main.cpp)
target_link_libraries(superjupiter-threshold-events-consumer PRIVATE SuperJupiter::ThresholdEvents)
